int n; // number of vertices
vector<vector<int>> adj; // adjacency list of graph
vector<bool> visited;
vector<int> ans;

void dfs(int v) {
    visited[v] = true;
    for (int u : adj[v]) {
        if (!visited[u]) {
            dfs(u);
        }
    }
    ans.push_back(v);
}

void topological_sort() {
    visited.assign(n, false);
    ans.clear();
    for (int i = 0; i < n; ++i) {
        if (!visited[i]) {
            dfs(i);
        }
    }
    reverse(ans.begin(), ans.end());
}

            q.pop();

                if (side[u] == -1) {

                    q.push(u);

        }
    }

const int INF = 1000000000;
vector<vector<pair<int, int>>> adj;

void dijkstra(int s, vector<int> & d, vector<int> & p) {
    int n = adj.size();
    d.assign(n, INF);
    p.assign(n, -1);
    vector<bool> u(n, false);

    d[s] = 0;
    for (int i = 0; i < n; i++) {
        int v = -1;
        for (int j = 0; j < n; j++) {
            if (!u[j] && (v == -1 || d[j] < d[v]))
                v = j;
        }

        if (d[v] == INF)
            break;

        u[v] = true;
        for (auto edge : adj[v]) {
            int to = edge.first;
            int len = edge.second;

            if (d[v] + len < d[to]) {
                d[to] = d[v] + len;
                p[to] = v;
            }
        }
    }
}

Cycle detection directed graph
int n;
vector<vector<int>> adj;
vector<char> color;
vector<int> parent;
int cycle_start, cycle_end;

bool dfs(int v) {
    color[v] = 1;
    for (int u : adj[v]) {
        if (color[u] == 0) {
            parent[u] = v;
            if (dfs(u))
                return true;
        } else if (color[u] == 1) {
            cycle_end = v;
            cycle_start = u;
            return true;
        }
    }
    color[v] = 2;
    return false;
}

void find_cycle() {
    color.assign(n, 0);
    parent.assign(n, -1);
    cycle_start = -1;

    for (int v = 0; v < n; v++) {
        if (color[v] == 0 && dfs(v))
            break;
    }

    if (cycle_start == -1) {
        cout << "Acyclic" << endl;
    } else {
        vector<int> cycle;
        cycle.push_back(cycle_start);
        for (int v = cycle_end; v != cycle_start; v = parent[v])
            cycle.push_back(v);
        cycle.push_back(cycle_start);
        reverse(cycle.begin(), cycle.end());

        cout << "Cycle found: ";
        for (int v : cycle)
            cout << v << " ";
        cout << endl;
    }
}

Cycle detection undirected graph
int n;
vector<vector<int>> adj;
vector<bool> visited;
vector<int> parent;
int cycle_start, cycle_end;

bool dfs(int v, int par) { // passing vertex and its parent vertex
    visited[v] = true;
    for (int u : adj[v]) {
        if(u == par) continue; // skipping edge to parent vertex
        if (visited[u]) {
            cycle_end = v;
            cycle_start = u;
            return true;
        }
        parent[u] = v;
        if (dfs(u, parent[u]))
            return true;
    }
    return false;
}

void find_cycle() {
    visited.assign(n, false);
    parent.assign(n, -1);
    cycle_start = -1;

    for (int v = 0; v < n; v++) {
        if (!visited[v] && dfs(v, parent[v]))
            break;
    }

    if (cycle_start == -1) {
        cout << "Acyclic" << endl;
    } else {
        vector<int> cycle;
        cycle.push_back(cycle_start);
        for (int v = cycle_end; v != cycle_start; v = parent[v])
            cycle.push_back(v);
        cycle.push_back(cycle_start);

        cout << "Cycle found: ";
        for (int v : cycle)
            cout << v << " ";
        cout << endl;
    }
}
MST (Prims)
#include <iostream>
#include <vector>
#include <queue>
#include <climits>

using namespace std;

// Pair representing (weight, destination_node)
typedef pair<int, int> pii;

void primsMST(vector<vector<pii>>& adj, int startNode, int V) {
    // Min-heap to pick the minimum weight edge
    priority_queue<pii, vector<pii>, greater<pii>> pq;

    vector<int> key(V, INT_MAX);   // Minimum weight to connect node
    vector<int> parent(V, -1);    // To store the structure of MST
    vector<bool> inMST(V, false); // To track nodes included in MST

    // Start with the initial node
    pq.push({0, startNode});
    key[startNode] = 0;

    int totalCost = 0;

    while (!pq.empty()) {
        int u = pq.top().second;
        int weight = pq.top().first;
        pq.pop();

        if (inMST[u]) continue; // Skip if node is already in MST

        inMST[u] = true;
        totalCost += weight;

        // Explore neighbors
        for (auto& neighbor : adj[u]) {
            int v = neighbor.first;
            int w = neighbor.second;

            // If v is not in MST and weight of (u,v) is smaller than current key
            if (!inMST[v] && w < key[v]) {
                key[v] = w;
                pq.push({key[v], v});
                parent[v] = u;
            }
        }
    }

    // Print the result
    cout << "Edge \tWeight" << endl;
    for (int i = 1; i < V; ++i) {
        if (parent[i] != -1)
            cout << parent[i] << " - " << i << " \t" << key[i] << endl;
    }
    cout << "Total Minimum Cost: " << totalCost << endl;
}

int main() {
    int V = 6; // Number of vertices
    vector<vector<pii>> adj(V);

    // Adding edges: adj[u].push_back({v, weight})
    adj[0].push_back({1, 2}); adj[1].push_back({0, 2});

    primsMST(adj, 0, V);

}

Investigation (Dijkstra with states)
#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
 
using namespace std;
 
#define ll long long
 
static const ll MOD = 1e9 + 7;
const ll INF = 1e17;
 
ll dist[100005];    
ll ways[100005];    
ll min_f[100005];  
ll max_f[100005];
vector<vector<pair<ll, ll>>> adj;
 
void dijkstra(ll n) {
    // 1. Initialize all arrays
    for (int i = 1; i <= n; i++) {
        dist[i] = INF;
        ways[i] = 0;
        min_f[i] = INF;
        max_f[i] = 0;
    }
 
    // 2. Base Case: Start at City 1
    dist[1] = 0;
    ways[1] = 1;
    min_f[1] = 0;
    max_f[1] = 0;
 
    // pair<price, node>
    priority_queue<pair<ll, ll>, vector<pair<ll, ll>>, greater<pair<ll, ll>>> q;
    q.push({0, 1});
 
    while (!q.empty()) {
        ll d_u = q.top().first;
        ll u = q.top().second;
        q.pop();
 
        // 3. Standard Dijkstra check
        if (d_u > dist[u]) continue;
 
        for (auto& edge : adj[u]) {
            ll v = edge.first;
            ll weight = edge.second;
 
            // Case A: Found a strictly better price
            if (dist[u] + weight < dist[v]) {
                dist[v] = dist[u] + weight;
                ways[v] = ways[u];
                min_f[v] = min_f[u] + 1;
                max_f[v] = max_f[u] + 1;
                q.push({dist[v], v});
            } 
            // Case B: Found an alternative path with the same minimum price
            else if (dist[u] + weight == dist[v]) {
                ways[v] = (ways[v] + ways[u]) % MOD;
                min_f[v] = min(min_f[v], min_f[u] + 1);
                max_f[v] = max(max_f[v], max_f[u] + 1);
            }
        }
    }
    
    cout << dist[n] << " " << ways[n] << " " << min_f[n] << " " << max_f[n] << endl;
}
 
void solve() {
    ll n, m;
    if (!(cin >> n >> m)) return;
    
    adj.assign(n + 1, vector<pair<ll, ll>>());
    for (ll i = 0; i < m; i++) {
        ll a, b, c;
        cin >> a >> b >> c;
        adj[a].push_back({b, c});
    }
 
    dijkstra(n);
}
 
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    // precompute(); // Not needed for this specific logic, but can keep if used elsewhere
    solve();
    return 0;
}
